#+TITLE: loader — how a profile is assembled
#+PROPERTY: header-args:emacs-lisp :tangle ./loader.el
#+PROPERTY: header-args+: :results none

* Why this is shared

The three profile directories each contain a ten-line bootstrap and nothing
else. The logic that assembles a profile lives here, once, so a change to how
manifests are read cannot drift between profiles.

A profile is: this repository's =base/=, then the profile's own =modules/=,
each in the order its =MANIFEST= declares. Nothing else decides load order —
not filename prefixes, not directory listing order, not "later files win".
That last habit is what made the previous configuration impossible to reason
about.

* Priced

~0.4 ms: three small files, one =re-search-forward= per non-comment line.

#+begin_src emacs-lisp
;; -*- lexical-binding: t; -*-
(require 'subr-x)

;; Bound by the ten-line bootstrap in each profile directory.
(defvar my/config-dir)
(defvar my/repo-root)
(defvar my/profile)

(defvar my/load-times nil
  "Modules loaded, as (filename . milliseconds); filled only under MY_BENCH.")
(declare-function my/mark "bench" (label))
(declare-function my/report "bench" ())

(defun my/load (file)
  "Load FILE, recording how long it took when bench.el is loaded."
  (if (boundp 'my/marks)
      (let ((t0 (float-time)))
        (load file nil 'nomessage)
        (push (cons (file-name-nondirectory file) (* 1000 (- (float-time) t0)))
              my/load-times))
    (load file nil 'nomessage)))

(defun my/bench-mark (label)
  "Record a milestone for bin/bench.sh, or do nothing in a normal run."
  (when (boundp 'my/marks)
    (my/mark label)))

(defun my/load-manifest (dir)
  "Load the files named in DIR/MANIFEST, in the order listed.

A missing MANIFEST is not an error: it means the directory has nothing to
contribute. A missing *entry* is an error, and so is an entry that resolves to
nothing — the first version of this function matched only lines starting with a
letter, digit or underscore, so an entry beginning with a dot-slash made it
vanish without a word. A module silently not loading is the worst possible
failure mode here. (This paragraph used to contain a bare quoted dot-slash,
which closed the docstring string and turned the rest into code — the same
mistake as an unescaped quote anywhere in Elisp.)"
  (let ((manifest (expand-file-name "MANIFEST" dir)))
    (when (file-readable-p manifest)
      (with-temp-buffer
        (insert-file-contents manifest)
        (goto-char (point-min))
        (while (not (eobp))
          (let ((line (string-trim (buffer-substring (line-beginning-position)
                                                     (line-end-position)))))
            (unless (or (string-empty-p line) (string-prefix-p "#" line))
              (let ((file (expand-file-name line dir)))
                (unless (file-readable-p file)
                  (error "MANIFEST entry not readable: %s (in %s)" line manifest))
                (my/load file))))
          (forward-line 1))))))

(my/bench-mark "1 early-init done")

(my/load-manifest (expand-file-name "base" my/repo-root))
(my/bench-mark "2 base modules done")

(my/load-manifest (expand-file-name "modules" my/config-dir))
(my/bench-mark "3 profile modules done")

(when (getenv "MY_BENCH")
  (my/report))
#+end_src
